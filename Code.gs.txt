/**
 * OLYMPUS ROADMAP ENGINE - PHASE C BACKEND (MULTIPLEXED)
 * Target Read Databases: "Mission_Log" & "Knowledge_Garden"
 * Target Write Database: "Mission_Log"
 */

// 1. READ PIPELINE (Serves Unified Data to Dashboard)
function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var jsonArray = [];

    // --- Extract Tab 1: Mission_Log ---
    var missionSheet = ss.getSheetByName("Mission_Log");
    if (missionSheet) {
      var mData = missionSheet.getDataRange().getValues();
      var mHeaders = mData[0];
      for (var i = 1; i < mData.length; i++) {
        var row = mData[i];
        var record = {};
        var hasData = false;
        for (var j = 0; j < mHeaders.length; j++) {
          record[mHeaders[j]] = row[j];
          if (row[j] !== "") hasData = true; 
        }
        if (hasData) jsonArray.push(record);
      }
    }

    // --- Extract Tab 2: Knowledge_Garden ---
    var knowledgeSheet = ss.getSheetByName("Knowledge_Garden");
    if (knowledgeSheet) {
      var kData = knowledgeSheet.getDataRange().getValues();
      var kHeaders = kData[0];
      for (var i = 1; i < kData.length; i++) {
        var row = kData[i];
        var record = {};
        var hasData = false;
        for (var j = 0; j < kHeaders.length; j++) {
          record[kHeaders[j]] = row[j];
          if (row[j] !== "") hasData = true; 
        }
        if (hasData) jsonArray.push(record);
      }
    }

    return ContentService.createTextOutput(JSON.stringify(jsonArray))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      "status": "error",
      "message": "GET Halt: " + error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * HARDENED PHASE C TRANSACTION RECEIVER
 * Enforces Option B execution semantics over Cross-Origin Redirects.
 */
function doPost(e) {
  // Enforce global execution block to avoid race conditions across parallel browser requests
  var executionLock = LockService.getScriptLock();
  try {
    // Attempt to acquire a exclusive write lock for up to 30 seconds
    executionLock.waitLock(30000);
  } catch (lockException) {
    return ContentService.createTextOutput(JSON.stringify({
      "status": "error",
      "error_type": "LOCK_TIMEOUT",
      "message": "System busy. Execution lock acquisition timed out after 30 seconds."
    })).setMimeType(ContentService.MimeType.JSON);
  }

  try {
    // Structural Guard: Validate input existence
    if (!e || !e.postData || !e.postData.contents) {
      throw new Error("Payload Error: Missing text stream body inside incoming POST request.");
    }

    // Extract transaction packets from text stream data
    var batchData = JSON.parse(e.postData.contents);
    
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Mission_Log");
    if (!sheet) {
      throw new Error("Database Error: 'Mission_Log' sheet tab could not be resolved.");
    }

    var dataMatrix = sheet.getDataRange().getValues();
    var headers = dataMatrix[0];
    
    // Resolve dynamic column positions to eliminate schema drift risks
    var idIdx = headers.indexOf("MISSION_ID");
    var statusIdx = headers.indexOf("STATUS");
    var completionDateIdx = headers.indexOf("COMPLETION_DATE");

    if (idIdx === -1 || statusIdx === -1) {
      throw new Error("Schema Violation: Critical header keys (MISSION_ID / STATUS) are missing.");
    }

    var mutationCount = 0;
    var systemTimestamp = new Date();

    // Execute sequential delta matching
    batchData.forEach(function(packet) {
      for (var rowIdx = 1; rowIdx < dataMatrix.length; rowIdx++) {
        if (dataMatrix[rowIdx][idIdx].toString() === packet.id.toString()) {
          var realSpreadsheetRow = rowIdx + 1; // Translate matrix index to physical row coordinate
          
          // Execute atomic state write
          sheet.getRange(realSpreadsheetRow, statusIdx + 1).setValue(packet.status);
          
          // Handle structural logging parameters
          if (completionDateIdx !== -1) {
            if (packet.status === "Completed") {
              sheet.getRange(realSpreadsheetRow, completionDateIdx + 1).setValue(systemTimestamp);
            } else {
              sheet.getRange(realSpreadsheetRow, completionDateIdx + 1).setValue(""); 
            }
          }
          mutationCount++;
          break; // Break loop layer for this resolved packet
        }
      }
    });

    // Force flush changes into underlying spreadsheet cells
    SpreadsheetApp.flush();
    executionLock.releaseLock();

    // Construct valid cross-origin compliance payload
    return ContentService.createTextOutput(JSON.stringify({
      "status": "success",
      "message": "Delta match successfully applied to " + mutationCount + " records."
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (runtimeException) {
    // Release execution lock safely during internal runtime failures
    if (executionLock.hasLock()) {
      executionLock.releaseLock();
    }
    
    // Fallback JSON block preventing browser-level CORS masking errors
    return ContentService.createTextOutput(JSON.stringify({
      "status": "error",
      "error_type": "RUNTIME_FAILURE",
      "message": runtimeException.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
